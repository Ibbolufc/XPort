# XPort core: the shared Remote Play library (lib/) and its dependencies, built for UWP.
#
# Visual Studio's UWP projects only support the MSVC (cl.exe) toolset, while lib/ and its
# dependencies are validated with clang-cl. So the core is built here, separately, with
# Ninja + clang-cl against the UWP API partition (WINAPI_FAMILY_APP), and the Xbox app
# imports the result through the generated xport-core.cmake: static libraries, include
# directories and the exact interface compile definitions the lib was built with (the app
# must see the same ChiakiSession layout; see lib/include/chiaki/session.h).
#
# Run from a Visual Studio developer environment (xbox/build.ps1 does all of this):
#   cmake -S xbox/core -B build-xbox/core -G Ninja -DCMAKE_BUILD_TYPE=Release `
#         -DCMAKE_C_COMPILER=clang-cl -DCMAKE_CXX_COMPILER=clang-cl
#   cmake --build build-xbox/core

cmake_minimum_required(VERSION 3.20)
project(xport-core C CXX)

if(NOT MSVC OR NOT CMAKE_C_COMPILER_ID STREQUAL "Clang")
	message(FATAL_ERROR "xbox/core must be built with clang-cl (use xbox/build.ps1)")
endif()

# Tells lib/ and third-party/ CMake that the target is UWP (Xbox)
set(CHIAKI_TARGET_UWP ON)
set(CHIAKI_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(CHIAKI_ENABLE_CLI OFF CACHE BOOL "" FORCE)
set(CHIAKI_ENABLE_FFMPEG_DECODER OFF CACHE STRING "" FORCE) # video decoding lands in Phase 4
set(CHIAKI_ENABLE_PI_DECODER OFF CACHE STRING "" FORCE)
set(CHIAKI_LIB_FETCH_DEPS ON CACHE BOOL "" FORCE)
set(CHIAKI_LIB_ENABLE_MBEDTLS ON CACHE BOOL "" FORCE)
set(CHIAKI_LIB_MBEDTLS_EXTERNAL_PROJECT ON CACHE BOOL "" FORCE)

# Compile against the UWP API partition, so any desktop-only Win32 call in lib/ or a
# dependency fails at compile time instead of at load time on the console.
set(XPORT_CORE_UWP_DEFINITIONS WINAPI_FAMILY=WINAPI_FAMILY_APP _WIN32_WINNT=0x0A00)
add_compile_definitions(${XPORT_CORE_UWP_DEFINITIONS})

add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../.." xport)

# Walk chiaki-lib's link graph and collect every static library (plus the interface include
# directories and compile definitions of everything on it).
function(_xport_walk item)
	string(REGEX REPLACE "^\\$<LINK_ONLY:(.*)>$" "\\1" item "${item}")
	if(NOT TARGET "${item}")
		return() # system library (ws2_32, bcrypt, ...): provided by WindowsApp.lib in the app
	endif()
	get_target_property(aliased "${item}" ALIASED_TARGET)
	if(aliased)
		set(item "${aliased}")
	endif()
	get_property(seen GLOBAL PROPERTY XPORT_CORE_SEEN)
	if("${item}" IN_LIST seen)
		return()
	endif()
	set_property(GLOBAL APPEND PROPERTY XPORT_CORE_SEEN "${item}")
	set(deps)
	get_target_property(type "${item}" TYPE)
	if(type STREQUAL "STATIC_LIBRARY")
		set_property(GLOBAL APPEND PROPERTY XPORT_CORE_STATIC "${item}")
		get_target_property(link_libs "${item}" LINK_LIBRARIES)
		if(link_libs)
			list(APPEND deps ${link_libs})
		endif()
	endif()
	get_target_property(iface_libs "${item}" INTERFACE_LINK_LIBRARIES)
	if(iface_libs)
		list(APPEND deps ${iface_libs})
	endif()
	foreach(dep ${deps})
		_xport_walk("${dep}")
	endforeach()
endfunction()
_xport_walk(chiaki-lib)
get_property(_xport_static_targets GLOBAL PROPERTY XPORT_CORE_STATIC)
get_property(_xport_seen GLOBAL PROPERTY XPORT_CORE_SEEN)
message(STATUS "XPort core static libraries: ${_xport_static_targets}")

set(_libs)
set(_incs)
set(_defs)
foreach(_t ${_xport_static_targets})
	list(APPEND _libs "$<TARGET_LINKER_FILE:${_t}>")
endforeach()
foreach(_t ${_xport_seen})
	list(APPEND _incs "$<TARGET_PROPERTY:${_t},INTERFACE_INCLUDE_DIRECTORIES>")
	list(APPEND _defs "$<TARGET_PROPERTY:${_t},INTERFACE_COMPILE_DEFINITIONS>")
endforeach()

file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/xport-core.cmake" CONTENT
"# Generated by xbox/core/CMakeLists.txt; do not edit.
set(XPORT_CORE_LIBRARIES \"$<JOIN:${_libs},;>\")
set(XPORT_CORE_INCLUDE_DIRS \"$<JOIN:$<REMOVE_DUPLICATES:$<FILTER:${_incs},EXCLUDE,^$>>,;>\")
set(XPORT_CORE_DEFINITIONS \"$<JOIN:$<REMOVE_DUPLICATES:${_defs};${XPORT_CORE_UWP_DEFINITIONS}>,;>\")
")
